# ---------------------------------------------------------------------------
# Minimal Python dependencies for running AlphaFlow on a single Blackwell GPU
# (e.g. RTX 5090, sm_120) with a plain `venv`.
#
# PyTorch itself is NOT listed here on purpose: it must come from the cu128
# wheel index so that the sm_120 kernels are available.  Install it first:
#
#   pip install torch==2.7.1 torchvision==0.22.1 \
#       --index-url https://download.pytorch.org/whl/cu128
#
# then:
#
#   pip install -r requirements-blackwell.txt
#
# (scripts/setup_env.sh does both automatically.)
# ---------------------------------------------------------------------------

# Core numerics / torch ecosystem
numpy<2
# scipy 1.10.1 is the version pinned by the original environment; newer 1.13+
# may compute FID-128 slightly wrong (sqrtm on degenerate matrices).
scipy==1.10.1; python_version < "3.12"
scipy>=1.11,<1.14; python_version >= "3.12"
Pillow
tqdm
einops
beartype
loguru

# Hydra config system + experiment glue
hydra-core==1.3.2
omegaconf==2.3.0
click
PyYAML
requests
GitPython

# Logging / monitoring (wandb is imported unconditionally by the code)
wandb>=0.17,<0.21
psutil
termcolor
matplotlib

# Training dependencies
torchmetrics==1.3.2
lpips==0.1.4
timm==1.0.11
fvcore
torch_dct
joblib==1.2.0

# Data / video decoding (imported at module import time)
av
ftfy
regex
protobuf

# Latent-space models (sd-vae) depend on HuggingFace diffusers.
diffusers>=0.32.2
accelerate
huggingface_hub
